<script>
    let db = JSON.parse(localStorage.getItem('NaimiPWA_DB')) || {};

    function getStorageKey() {
        const client = document.getElementById('clientName').value.trim() || 'عام';
        return `${client}_${document.getElementById('month').value}_${document.getElementById('year').value}`;
    }

    // جلب بيانات يوم معين بمجرد كتابة رقمه (للتعديل)
    function fetchDayData() {
        const key = getStorageKey();
        const day = document.getElementById('day').value;
        if(db[key] && db[key].days[day]) {
            const d = db[key].days[day];
            document.getElementById('bc').value = d.bc; document.getElementById('bp').value = d.bp;
            document.getElementById('lc').value = d.lc; document.getElementById('lp').value = d.lp;
            document.getElementById('dc').value = d.dc; document.getElementById('dp').value = d.dp;
        } else {
            // مسح الخانات إذا اليوم غير موجود للبدء بجديد
            ['bc','bp','lc','lp','dc','dp'].forEach(id => document.getElementById(id).value = '');
        }
    }

    function loadData() {
        const key = getStorageKey();
        const data = db[key] || { info: { acc: '', notes: '' }, days: {} };
        
        document.getElementById('accNum').value = data.info.acc || '';
        document.getElementById('notes').value = data.info.notes || '';
        document.getElementById('dispClient').innerText = document.getElementById('clientName').value || "غير محدد";
        document.getElementById('dispAcc').innerText = data.info.acc || "-";

        let tbody = document.getElementById('tbody');
        tbody.innerHTML = '';
        let total = 0, h1 = 0, h2 = 0;

        Object.keys(data.days).sort((a,b)=>a-b).forEach(day => {
            let d = data.days[day];
            let sum = (d.bc*d.bp) + (d.lc*d.lp) + (d.dc*d.dp);
            total += sum;
            if(parseInt(day) <= 15) h1 += sum; else h2 += sum;
            
            tbody.innerHTML += `<tr class="row-edit" onclick="setDay(${day})">
                <td>${day}</td><td>${d.bc}×${d.bp}</td><td>${d.lc}×${d.lp}</td><td>${d.dc}×${d.dp}</td><td>${sum.toLocaleString()}</td>
            </tr>`;
        });

        document.getElementById('grandTotal').innerText = total.toLocaleString() + " د.ع";
        document.getElementById('h1Total').innerText = h1.toLocaleString() + " د.ع";
        document.getElementById('h2Total').innerText = h2.toLocaleString() + " د.ع";
        document.getElementById('totalWords').innerText = tafeet(total);
    }

    function setDay(d) { document.getElementById('day').value = d; fetchDayData(); }

    function saveData() {
        const key = getStorageKey();
        const day = document.getElementById('day').value;
        if(!day) return alert("حدد اليوم أولاً");

        if(!db[key]) db[key] = { info: { acc: '', notes: '' }, days: {} };
        db[key].days[day] = {
            bc: Number(document.getElementById('bc').value), bp: Number(document.getElementById('bp').value),
            lc: Number(document.getElementById('lc').value), lp: Number(document.getElementById('lp').value),
            dc: Number(document.getElementById('dc').value), dp: Number(document.getElementById('dp').value)
        };
        saveToLocalStorage();
    }

    function saveInfo() {
        const key = getStorageKey();
        if(!db[key]) db[key] = { info: { acc: '', notes: '' }, days: {} };
        db[key].info.acc = document.getElementById('accNum').value;
        db[key].info.notes = document.getElementById('notes').value;
        saveToLocalStorage();
    }

    function saveToLocalStorage() {
        localStorage.setItem('NaimiPWA_DB', JSON.stringify(db));
        loadData();
    }

    function exportToExcel() {
        const client = document.getElementById('clientName').value || 'حسابات';
        const html = `<html><head><meta charset="utf-8"></head><body dir="rtl">
            <h2>${client}</h2>${document.getElementById('printableArea').innerHTML}
        </body></html>`;
        const blob = new Blob([html], { type: 'application/vnd.ms-excel' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a"); a.href = url; a.download = `${client}.xls`; a.click();
    }

    function clearMonth() { if(confirm("مسح بيانات الشهر؟")) { delete db[getStorageKey()]; saveToLocalStorage(); } }

    function tafeet(n) {
        if (n === 0) return "فقط صفر دينار لا غير";
        const a = ["", "واحد", "اثنان", "ثلاثة", "أربعة", "خمسة", "ستة", "سبعة", "ثمانية", "تسعة", "عشرة", "أحد عشر", "اثنا عشر", "ثلاثة عشر", "أربعة عشر", "خمسة عشر", "ستة عشر", "سبعة عشر", "ثمانية عشر", "تسعة عشر"];
        const b = ["", "", "عشرون", "ثلاثون", "أربعون", "خمسون", "ستون", "سبعون", "ثمانون", "تسعون"];
        const c = ["", "مائة", "مائتان", "ثلاثمائة", "أربعمائة", "خمسمائة", "ستمائة", "سبعمائة", "ثمانمائة", "تسعمائة"];
        function conv(n) {
            if (n < 20) return a[n];
            if (n < 100) return b[Math.floor(n / 10)] + (n % 10 !== 0 ? " و" + a[n % 10] : "");
            if (n < 1000) return c[Math.floor(n / 100)] + (n % 100 !== 0 ? " و" + conv(n % 100) : "");
            if (n < 1000000) return conv(Math.floor(n / 1000)) + " ألف " + (n % 1000 !== 0 ? " و" + conv(n % 1000) : "");
            return conv(Math.floor(n / 1000000)) + " مليون " + (n % 1000000 !== 0 ? " و" + conv(n % 1000000) : "");
        }
        return "فقط " + conv(n) + " دينار عراقي لا غير";
    }

    window.onload = () => { 
        const d = new Date(); 
        document.getElementById('month').value = d.getMonth()+1; 
        loadData(); 
    };
</script>
